% Keep diagrams and other graphics compact while preserving their aspect ratio
% and preventing wide or tall figures from being cut off.
\setkeys{Gin}{width=\linewidth,keepaspectratio}
\let\HSOriginalIncludeGraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \adjustbox{max width=.8\linewidth,max height=.55\textheight,keepaspectratio}{%
    \HSOriginalIncludeGraphics[#1]{#2}%
  }%
}

% Tables are information-dense in this document; slightly smaller table text
% gives long cells room to wrap without changing the surrounding body text.
\AtBeginEnvironment{longtable}{\footnotesize}
\fvset{breaklines=true,breakanywhere=true}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{breaklines=true,breakanywhere=true}

\let\HSOldUnderscore\_
\def\_{\HSOldUnderscore\allowbreak}
\sloppy

% Some identifiers have no underscore near a useful break point (or none
% at all — e.g. DataCollection.CameraCheckup()), so the \_ trick above
% isn't enough on its own. seqsplit allows a break between any two
% characters as needed to fit the column; safe here because it operates
% on the macro argument directly, unlike \url-family commands which
% require raw, untokenized source text.
\let\HSOriginalTexttt\texttt
\AtBeginEnvironment{longtable}{\renewcommand{\texttt}[1]{\HSOriginalTexttt{\seqsplit{#1}}}}
\AtEndEnvironment{longtable}{\let\texttt\HSOriginalTexttt}

% Keep explicit document links visibly distinct while leaving TOC and other
% internal structural links black through the Quarto configuration.
\AtBeginDocument{%
  \let\HSOriginalHref\href
  \renewcommand{\href}[2]{\normalem\HSOriginalHref{#1}{\textcolor{blue}{\uline{#2}}}\normalem}%
}
